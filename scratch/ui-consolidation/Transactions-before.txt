import { useEffect, useState } from 'react';
import * as Crypto from 'expo-crypto';
import { Alert, Modal, TouchableWithoutFeedback } from 'react-native';
import { Pressable, ScrollView, Text, TextInput, View } from '../components/NativeAppearance';
import { AppIcon } from '../components/AppIcon';
import { Card, FinancialScreen, ui } from '../components/FinancialScreen';
import { SelectDropdown, type DropdownOption } from '../components/SelectDropdown';
import { formatCurrency, acceptMoneyEdit, type DebtItem } from '../onboarding-domain';
import {
  classifyCategoryDirection,
  calculateTransactionBalanceDelta,
  type LedgerTransaction,
  type TransactionCategory,
} from '../finance/ledger';
import { makeMoney, moneyFromMajor, moneyToMajor, parseMoney, addMoney, type Money } from '../finance/money';
import { currentHouseholdDate, currentHouseholdMonth, getNextMonthString, getPreviousMonthString, parseDateOnly, type DateOnly } from '../finance/calendar';
import { resolveRecurringBillDate, type RecurringBill } from '../finance/recurring-bills';
import { createDebtMinimumPaymentTransaction, findUnassignedPaymentsForObligation, getObligationPaymentState, getScheduledObligationRows, type PaymentObligation, type ScheduledObligationRow } from '../finance/payment-matching';
import { type ObligationSchedules } from '../finance/obligation-schedule';

const CATEGORY_CHOICES: { label: string; category: TransactionCategory; isDebt: boolean }[] = [
  { label: 'Payment', category: 'payment', isDebt: true },
  { label: 'Single purchase', category: 'purchase', isDebt: true },
  { label: 'Subscription', category: 'subscription', isDebt: true },
  { label: 'Utilities', category: 'utilities', isDebt: true },
  { label: 'Interest', category: 'interest', isDebt: true },
  { label: 'Fee / Annual Fee', category: 'fee', isDebt: true },
  { label: 'Refund', category: 'refund', isDebt: true },
  { label: 'Debit Bill (Checking)', category: 'debit_bill', isDebt: false },
  { label: 'Debit Spending (Checking)', category: 'debit_spending', isDebt: false },
];

export function TransactionsScreen({
  debts,
  transactions = [],
  currencySymbol = '$',
  currencyCode = 'USD',
  onSettings,
  onAddTransaction,
  onUpdateTransaction,
  onDeleteTransaction,
  recurringBills = [],
  obligationSchedules,
  loanBillPairs = {},
  householdTimeZone,
  onRecordBillPayment,
  initialTransactionId,
  initialSection = 'History',
  onEditDismiss,
  budgetScheduled = false,
  onBudgetSection,
}: {
  budgetScheduled?: boolean;
  onBudgetSection?: (section: string) => void;
  initialTransactionId?: string | null;
  initialSection?: string;
  onEditDismiss?: () => void;
  debts: DebtItem[];
  transactions?: LedgerTransaction[];
  currencySymbol?: string;
  currencyCode?: string;
  onSettings: () => void;
  onAddTransaction?: (tx: LedgerTransaction) => unknown;
  onUpdateTransaction?: (tx: LedgerTransaction) => unknown;
  onDeleteTransaction?: (txId: string) => unknown;
  recurringBills?: RecurringBill[];
  obligationSchedules?: ObligationSchedules;
  loanBillPairs?: Readonly<Record<string, string>>;
  householdTimeZone?: string;
  onRecordBillPayment?: (bill: RecurringBill, dueCycle: DateOnly, amount?: Money) => unknown;
}) {
  const [tab, setTab] = useState(initialSection);
  const [scheduledMonthOffset, setScheduledMonthOffset] = useState(0);
  const [accountIdx, setAccountIdx] = useState(0); // 0 = All accounts, 1 = Debit/Checking, 2..N = debt accounts
  const [periodIdx, setPeriodIdx] = useState(0); // 0 = Current month, 1 = Previous month, 2 = All time
  const [typeIdx, setTypeIdx] = useState(0); // 0 = All Types, 1 = Payments, 2 = Charges/Fees, 3 = Debit Only
  const [sortAsc, setSortAsc] = useState(false); // false = newest first, true = oldest first
  const [search, setSearch] = useState('');
  const [showAddForm, setShowAddForm] = useState(false);

  // Add Transaction Form State
  const [newPayee, setNewPayee] = useState('');
  const [newAmount, setNewAmount] = useState('');
  const [newCategoryIdx, setNewCategoryIdx] = useState(0);
  const [newAccountId, setNewAccountId] = useState(debts[0]?.id ?? '');
  const [newDate, setNewDate] = useState(() => currentHouseholdDate(householdTimeZone));
  const [newAppliesToDate, setNewAppliesToDate] = useState('');
  const [newPrincipal, setNewPrincipal] = useState('');
  const [newInterest, setNewInterest] = useState('');
  const [newEscrow, setNewEscrow] = useState('');
  const [newFee, setNewFee] = useState('');
  const [savingTx, setSavingTx] = useState(false);
  const [showDatePicker, setShowDatePicker] = useState(false);
  const [datePickerMonth, setDatePickerMonth] = useState(() => currentHouseholdMonth(householdTimeZone));
  const [formError, setFormError] = useState('');
  const [billError, setBillError] = useState('');
  const [pendingBillCycle, setPendingBillCycle] = useState<string | null>(null);
  const [upcomingTarget, setUpcomingTarget] = useState<ScheduledObligationRow | null>(null);
  const [upcomingBusy, setUpcomingBusy] = useState(false);
  const [upcomingError, setUpcomingError] = useState('');

  // Edit Transaction Form State
  const [editingTx, setEditingTx] = useState<LedgerTransaction | null>(null);
  const [editPayee, setEditPayee] = useState('');
  const [editAmount, setEditAmount] = useState('');
  const [editCategoryIdx, setEditCategoryIdx] = useState(0);
  const [editAccountId, setEditAccountId] = useState<string | null>(null);
  const [editDate, setEditDate] = useState('');
  const [editMemo, setEditMemo] = useState('');
  const [editPrincipal, setEditPrincipal] = useState('');
  const [editInterest, setEditInterest] = useState('');
  const [editEscrow, setEditEscrow] = useState('');
  const [editFee, setEditFee] = useState('');
  const [editError, setEditError] = useState('');
  const [savingEdit, setSavingEdit] = useState(false);
  const [deletingTx, setDeletingTx] = useState(false);

  const todayStr = currentHouseholdDate(householdTimeZone);
  const currentMonthStr = currentHouseholdMonth(householdTimeZone);
  const prevMonthStr = getPreviousMonthString(currentMonthStr);

  const accountOptions = [
    { label: 'All accounts', id: 'all' },
    { label: 'Checking / Debit ($0 debt)', id: 'debit' },
    ...debts.map(d => ({ label: d.name, id: d.id })),
  ];

  const periodDropdownOptions: readonly DropdownOption<number>[] = [
    { value: 0, label: 'Current month', icon: 'calendar', subtitle: 'View transactions for this month' },
    { value: 1, label: 'Previous month', icon: 'calendar', subtitle: 'View transactions from last month' },
    { value: 2, label: 'All time', icon: 'calendar', subtitle: 'Lifetime transaction history' },
  ];
  const typeDropdownOptions: readonly DropdownOption<number>[] = [
    { value: 0, label: 'All types' }, { value: 1, label: 'Payments' },
    { value: 2, label: 'Charges and fees' }, { value: 3, label: 'Debit only' },
  ];
  const sortDropdownOptions: readonly DropdownOption<number>[] = [
    { value: 0, label: 'Newest first', icon: 'sort' }, { value: 1, label: 'Oldest first', icon: 'sort' },
  ];

  const accountFilterDropdownOptions: readonly DropdownOption<number>[] = accountOptions.map((opt, idx) => ({
    value: idx,
    label: opt.label,
    icon: idx === 1 ? 'calendar' : (idx > 1 ? 'creditCard' : undefined),
  }));

  const categoryDropdownOptions: readonly DropdownOption<number>[] = CATEGORY_CHOICES.map((c, idx) => ({
    value: idx,
    label: c.label,
    subtitle: c.isDebt ? 'Debt account transaction' : 'Checking cashflow ($0 debt)',
    icon: c.category === 'payment' ? 'check' : (c.isDebt ? 'creditCard' : 'calendar'),
  }));

  const debtAccountDropdownOptions: readonly DropdownOption<string>[] = debts.map(d => ({
    value: d.id,
    label: d.name,
    subtitle: `Balance: ${formatCurrency(d.balance, currencySymbol)}`,
    icon: 'creditCard',
  }));

  const selectedDebtId = newAccountId || debts[0]?.id || '';
  const selectedDebt = debts.find(debt => debt.id === selectedDebtId);
  const dueCycleOptions: readonly DropdownOption<string>[] = [
    { value: '', label: 'No due cycle', subtitle: 'Leave this payment unassigned to a due date' },
    ...(selectedDebt
      ? [0, 1, 2].map(monthOffset => {
          let month = currentMonthStr;
          for (let offset = 0; offset < monthOffset; offset++) month = getNextMonthString(month);
          const cycleDay = selectedDebt.dueDay ?? Number(newDate.slice(-2));
          const date = resolveRecurringBillDate(month, cycleDay);
          const labelDate = new Date(`${date}T12:00:00`);
          const label = new Intl.DateTimeFormat(undefined, { month: 'short', day: 'numeric', year: 'numeric' }).format(labelDate);
          return { value: date, label: `Due ${label}`, subtitle: selectedDebt.dueDay ? `Apply this payment to ${label}` : `Link this payment to the ${label} cycle` };
        })
      : []),
  ];

  const upcomingObligation: PaymentObligation | null = upcomingTarget ? {
    id: upcomingTarget.kind === 'debt' ? `minimum:${upcomingTarget.debt?.id ?? upcomingTarget.id}` : upcomingTarget.bill?.id ?? upcomingTarget.id,
    kind: upcomingTarget.kind,
    name: upcomingTarget.name,
    dueDate: upcomingTarget.dueDate,
    dueCycleMonth: upcomingTarget.dueCycleMonth,
    requiredAmount: upcomingTarget.requiredAmount,
    accountId: upcomingTarget.kind === 'debt' ? upcomingTarget.debt?.id : upcomingTarget.bill?.creditAccountId,
    billId: upcomingTarget.bill?.id,
    bill: upcomingTarget.bill,
    debt: upcomingTarget.debt,
  } : null;
  const upcomingPaymentState = upcomingObligation ? getObligationPaymentState(upcomingObligation, transactions, currencyCode) : null;
  const upcomingCandidates = upcomingObligation ? findUnassignedPaymentsForObligation(upcomingObligation, transactions) : [];

  const matchUpcomingPayment = async (tx: LedgerTransaction) => {
    if (!upcomingTarget || !upcomingObligation || !onUpdateTransaction || upcomingBusy) return;
    setUpcomingBusy(true); setUpcomingError('');
    try {
      const billId = upcomingTarget.bill?.id;
      const memo = billId ? (tx.memo?.includes(`[bill:${billId}]`) ? tx.memo : `${tx.memo ?? ''}${tx.memo ? ' ' : ''}[bill:${billId}]`) : tx.memo;
      await onUpdateTransaction({ ...tx, appliesToDate: upcomingTarget.dueDate, accountId: tx.accountId ?? upcomingObligation.accountId ?? null, memo });
      setUpcomingTarget(null);
    } catch (error) { setUpcomingError(error instanceof Error ? error.message : 'Unable to match this payment.'); }
    finally { setUpcomingBusy(false); }
  };

  const recordUpcomingPayment = async () => {
    if (!upcomingTarget || !upcomingPaymentState || upcomingBusy) return;
    const remaining = upcomingPaymentState.confirmedMatch.remainingAmount;
    if (remaining.minorUnits <= 0 || upcomingPaymentState.status === 'pending_confirmation') {
      setUpcomingError('This cycle is already satisfied or has a payment awaiting confirmation.');
      return;
    }
    setUpcomingBusy(true); setUpcomingError('');
    try {
      if (upcomingTarget.kind === 'debt' && upcomingTarget.debt && onAddTransaction) {
        await onAddTransaction(createDebtMinimumPaymentTransaction({ id: `tx:${Crypto.randomUUID()}`, debt: upcomingTarget.debt, today: todayStr, dueDate: upcomingTarget.dueDate, amount: remaining, createdAt: new Date().toISOString() }));
      } else if (upcomingTarget.kind === 'bill' && upcomingTarget.bill && onRecordBillPayment) {
        await onRecordBillPayment({ ...upcomingTarget.bill, amount: upcomingTarget.requiredAmount }, upcomingTarget.dueDate, remaining);
      }
      setUpcomingTarget(null);
    } catch (error) { setUpcomingError(error instanceof Error ? error.message : 'Unable to record this payment.'); }
    finally { setUpcomingBusy(false); }
  };

  // Filter transactions
  const filtered = transactions.filter(tx => {
    if (tx.deletedAt !== null) return false;

    // Period filter
    if (periodIdx === 0 && !tx.date.startsWith(currentMonthStr)) return false;
    if (periodIdx === 1 && !tx.date.startsWith(prevMonthStr)) return false;

    // Account filter
    const selectedAcc = accountOptions[accountIdx];
    if (selectedAcc.id === 'debit' && tx.accountId !== null) return false;
    if (selectedAcc.id !== 'all' && selectedAcc.id !== 'debit' && tx.accountId !== selectedAcc.id) return false;

    // Type filter
    const dir = classifyCategoryDirection(tx.category);
    if (typeIdx === 1 && dir !== 'decrease_debt') return false; // Payments / refunds
    if (typeIdx === 2 && dir !== 'increase_debt') return false; // Charges / interest / fees
    if (typeIdx === 3 && dir !== 'cash_only') return false; // Debit only

    // Search query
    if (search.trim()) {
      const q = search.toLowerCase();
      const matchPayee = tx.payee.toLowerCase().includes(q);
      const matchCategory = tx.category.toLowerCase().includes(q);
      const matchMemo = tx.memo ? tx.memo.toLowerCase().includes(q) : false;
      if (!matchPayee && !matchCategory && !matchMemo) return false;
    }

    return true;
  });

  // Sort transactions by date
  const sorted = [...filtered].sort((a, b) => {
    if (sortAsc) return a.date.localeCompare(b.date);
    return b.date.localeCompare(a.date);
  });

  // Calculate summary metrics for the filtered period
  let totalPaymentsMinor = 0;
  let totalInterestFeesMinor = 0;
  let totalAdjustmentsMinor = 0;
  let netBalanceChangeMinor = 0;

  for (const tx of filtered) {
    const dir = classifyCategoryDirection(tx.category);
    if (tx.category === 'payment' || tx.category === 'refund') {
      totalPaymentsMinor += tx.amount.minorUnits;
    }
    if (['interest', 'fee', 'annual_fee'].includes(tx.category)) {
      totalInterestFeesMinor += tx.amount.minorUnits;
    }
    if (['interest_adjustment', 'balance_adjustment', 'credit_adjustment'].includes(tx.category)) {
      totalAdjustmentsMinor += tx.amount.minorUnits;
    }
    if (dir === 'increase_debt') {
      netBalanceChangeMinor += tx.amount.minorUnits;
    } else if (dir === 'decrease_debt') {
      netBalanceChangeMinor -= tx.amount.minorUnits;
    }
  }

  const handleSaveTransaction = async () => {
    setFormError('');
    if (!newAmount || parseFloat(newAmount) <= 0) {
      setFormError('Please enter a valid amount');
      return;
    }

    const catOption = CATEGORY_CHOICES[newCategoryIdx];
    let accountId: string | null = null;

    if (catOption.isDebt) {
      if (!selectedDebt) {
        setFormError(debts.length === 0 ? 'Please add a debt account on the Debt tab first' : 'Select a debt account for this transaction');
        return;
      }
      accountId = selectedDebt.id;
    }

    setSavingTx(true);
    try {
      const actualDate = parseDateOnly(newDate.trim());
      const appliesToDate = newAppliesToDate.trim() ? parseDateOnly(newAppliesToDate.trim()) : null;
      const moneyAmount = parseMoney(newAmount, currencyCode);

      const principalAmount = newPrincipal.trim() ? parseMoney(newPrincipal.trim(), currencyCode) : null;
      const interestAmount = newInterest.trim() ? parseMoney(newInterest.trim(), currencyCode) : null;
      const escrowAmount = newEscrow.trim() ? parseMoney(newEscrow.trim(), currencyCode) : null;
      const feeAmount = newFee.trim() ? parseMoney(newFee.trim(), currencyCode) : null;

      const tx: LedgerTransaction = {
        id: `tx:${Crypto.randomUUID()}`,
        accountId,
        date: actualDate,
        appliesToDate,
        amount: moneyAmount,
        category: catOption.category,
        payee: newPayee.trim() || (catOption.category === 'payment' ? `Payment to ${selectedDebt?.name ?? 'debt account'}` : catOption.label),
        memo: `Manual entry (${catOption.label})`,
        source: 'manual',
        reconciled: false,
        createdAt: new Date().toISOString(),
        deletedAt: null,
        principalAmount,
        interestAmount,
        escrowAmount,
        feeAmount,
      };

      if (onAddTransaction) {
        const res = await onAddTransaction(tx);
        if (res && (res as { outcome?: string }).outcome === 'pending') {
          Alert.alert(
            'Saved Locally (Pending Confirmation)',
            'Transaction recorded locally. Confirmation will be verified upon connecting to the server.',
          );
        }
      }
      setShowAddForm(false);
      setNewPayee('');
      setNewAmount('');
      setNewAppliesToDate('');
      setNewPrincipal('');
      setNewInterest('');
      setNewEscrow('');
      setNewFee('');
    } catch (e) {
      const msg = (e as Error).message || 'Unable to save transaction';
      setFormError(msg);
      if (msg.includes('conflict') || msg.includes('Conflict')) {
        Alert.alert('Save Conflict', msg);
      }
    } finally {
      setSavingTx(false);
    }
  };

  const handleOpenEdit = (tx: LedgerTransaction) => {
    setEditingTx(tx);
    setEditPayee(tx.payee);
    setEditAmount(moneyToMajor(tx.amount).toFixed(2));
    const catIdx = CATEGORY_CHOICES.findIndex(c => c.category === tx.category);
    setEditCategoryIdx(catIdx >= 0 ? catIdx : 0);
    setEditAccountId(tx.accountId);
    setEditDate(tx.date);
    setEditMemo(tx.memo || '');
    setEditPrincipal(tx.principalAmount ? moneyToMajor(tx.principalAmount).toFixed(2) : '');
    setEditInterest(tx.interestAmount ? moneyToMajor(tx.interestAmount).toFixed(2) : '');
    setEditEscrow(tx.escrowAmount ? moneyToMajor(tx.escrowAmount).toFixed(2) : '');
    setEditFee(tx.feeAmount ? moneyToMajor(tx.feeAmount).toFixed(2) : '');
    setEditError('');
  };

  useEffect(() => {
    const target = transactions.find(tx => tx.id === initialTransactionId && tx.deletedAt === null);
    if (target) handleOpenEdit(target);
  // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [initialTransactionId]);

  useEffect(() => {
    setTab(initialSection);
    if (initialSection === 'Scheduled') setScheduledMonthOffset(0);
  }, [initialSection]);

  const handleSaveEdit = async () => {
    if (!editingTx || savingEdit) return;
    setEditError('');
    const parsedAmt = parseFloat(editAmount);
    if (Number.isNaN(parsedAmt) || parsedAmt <= 0) {
      setEditError('Please enter a valid amount');
      return;
    }
    const catOption = CATEGORY_CHOICES[editCategoryIdx];
    if (catOption.isDebt && !editAccountId) {
      setEditError(debts.length === 0 ? 'Please add a debt account on the Debt tab first' : 'Select a debt account for this transaction');
      return;
    }

    setSavingEdit(true);
    try {
      const actualDate = parseDateOnly(editDate.trim());
      const moneyAmount = parseMoney(editAmount, currencyCode);
      const principalAmount = editPrincipal.trim() ? parseMoney(editPrincipal.trim(), currencyCode) : null;
      const interestAmount = editInterest.trim() ? parseMoney(editInterest.trim(), currencyCode) : null;
      const escrowAmount = editEscrow.trim() ? parseMoney(editEscrow.trim(), currencyCode) : null;
      const feeAmount = editFee.trim() ? parseMoney(editFee.trim(), currencyCode) : null;

      const updatedTx: LedgerTransaction = {
        ...editingTx,
        accountId: catOption.isDebt ? editAccountId : null,
        date: actualDate,
        amount: moneyAmount,
        category: catOption.category,
        payee: editPayee.trim() || catOption.label,
        memo: editMemo.trim() || undefined,
        principalAmount,
        interestAmount,
        escrowAmount,
        feeAmount,
      };

      if (onUpdateTransaction) {
        const res = await onUpdateTransaction(updatedTx);
        if (res && (res as { outcome?: string }).outcome === 'pending') {
          Alert.alert(
            'Updated Locally (Pending Confirmation)',
            'Transaction update saved locally. Confirmation will be verified upon connecting to the server.',
          );
        }
      }
      setEditingTx(null); onEditDismiss?.();
    } catch (e) {
      const msg = (e as Error).message || 'Unable to save transaction';
      setEditError(msg);
      if (msg.includes('conflict') || msg.includes('Conflict')) {
        Alert.alert('Update Conflict', msg);
      }
    } finally {
      setSavingEdit(false);
    }
  };

  const handleDeleteTx = async () => {
    if (!editingTx || deletingTx) return;
    setDeletingTx(true);
    try {
      if (onDeleteTransaction) {
        const res = await onDeleteTransaction(editingTx.id);
        if (res && (res as { outcome?: string }).outcome === 'pending') {
          Alert.alert(
            'Deleted Locally (Pending Confirmation)',
            'Transaction deletion recorded locally. Confirmation will be verified upon connecting to the server.',
          );
        }
      } else if (onUpdateTransaction) {
        await onUpdateTransaction({
          ...editingTx,
          deletedAt: new Date().toISOString(),
        });
      }
      setEditingTx(null); onEditDismiss?.();
    } catch (e) {
      const msg = (e as Error).message || 'Unable to delete transaction';
      setEditError(msg);
      if (msg.includes('conflict') || msg.includes('Conflict')) {
        Alert.alert('Delete Conflict', msg);
      }
    } finally {
      setDeletingTx(false);
    }
  };

  // Live Balance Preview Calculation for Edit Modal
  const editCatOption = CATEGORY_CHOICES[editCategoryIdx];
  const targetEditDebt = debts.find(d => d.id === editAccountId);
  let editCurrentBal = targetEditDebt ? targetEditDebt.balance : 0;
  let editAfterBal = editCurrentBal;

  if (targetEditDebt) {
    const curMoney = moneyFromMajor(targetEditDebt.balance, currencyCode);
    let previewMoney = curMoney;

    // 1. Revert original transaction if it targeted this debt and was active
    if (editingTx && editingTx.accountId === targetEditDebt.id && editingTx.deletedAt === null) {
      const origDelta = calculateTransactionBalanceDelta(editingTx, targetEditDebt.category);
      previewMoney = addMoney(previewMoney, makeMoney(currencyCode, -origDelta.minorUnits));
    }

    // 2. Apply newly configured amount and category
    const parsedAmt = parseFloat(editAmount);
    if (!Number.isNaN(parsedAmt) && parsedAmt > 0 && editCatOption.isDebt) {
      try {
        const newAmtMoney = parseMoney(editAmount, currencyCode);
        const pAmt = editPrincipal.trim() ? parseMoney(editPrincipal.trim(), currencyCode) : null;
        const iAmt = editInterest.trim() ? parseMoney(editInterest.trim(), currencyCode) : null;
        const eAmt = editEscrow.trim() ? parseMoney(editEscrow.trim(), currencyCode) : null;
        const fAmt = editFee.trim() ? parseMoney(editFee.trim(), currencyCode) : null;

        const candidateTx: LedgerTransaction = {
          ...editingTx!,
          accountId: targetEditDebt.id,
          amount: newAmtMoney,
          category: editCatOption.category,
          principalAmount: pAmt,
          interestAmount: iAmt,
          escrowAmount: eAmt,
          feeAmount: fAmt,
        };
        const newDelta = calculateTransactionBalanceDelta(candidateTx, targetEditDebt.category);
        previewMoney = addMoney(previewMoney, newDelta);
      } catch {
        // invalid input ignored for preview
      }
    }

    editAfterBal = moneyToMajor(previewMoney);
  }

  return (
    <FinancialScreen
      title="Transactions"
      segments={['History', 'Upcoming', 'Scheduled']}
      selected={tab}
      onSelect={nextTab => {
        setTab(nextTab);
        if (nextTab !== 'History') setShowAddForm(false);
      }}
      onSettings={onSettings}
      onAdd={tab === 'History' && !showAddForm ? () => setShowAddForm(true) : undefined}
      addLabel="Add transaction"
    >
      {showAddForm ? (
        <Card>
          <View style={[ui.row, { justifyContent: 'space-between' }]}>
            <Text style={ui.heading}>Add Transaction</Text>
            <Pressable
              accessibilityRole="button"
              accessibilityLabel="Dismiss form"
              onPress={() => setShowAddForm(false)}
            >
              <Text style={ui.buttonText}>Cancel</Text>
            </Pressable>
          </View>

          {!!formError && (
            <Text style={[ui.small, { color: '#F08B78' }]}>{formError}</Text>
          )}

          <Text style={ui.small}>Payee or Description (optional)</Text>
          <TextInput
            placeholder={CATEGORY_CHOICES[newCategoryIdx].category === 'payment' ? 'Add a note (optional)' : 'e.g. Electric Company, Target'}
            placeholderTextColor="#AAB3AD"
            value={newPayee}
            onChangeText={setNewPayee}
            style={ui.input}
          />

          <Text style={ui.small}>Transaction date</Text>
          <Pressable
            accessibilityRole="button"
            accessibilityLabel={`Transaction date: ${newDate}`}
            onPress={() => { setDatePickerMonth(newDate.slice(0, 7)); setShowDatePicker(true); }}
            style={[ui.input, { flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }]}
          >
            <Text style={ui.text}>{newDate}</Text>
            <AppIcon name="calendar" size={18} color="#76DCB9" />
          </Pressable>
          {CATEGORY_CHOICES[newCategoryIdx].category === 'payment' ? (
            <SelectDropdown
              label="Applies to payment cycle (optional)"
              sheetTitle="Choose Payment Due Cycle"
              value={newAppliesToDate}
              options={dueCycleOptions}
              onSelect={setNewAppliesToDate}
              placeholder="Choose a due cycle"
            />
          ) : null}

          <Modal visible={showDatePicker} transparent animationType="slide" onRequestClose={() => setShowDatePicker(false)}>
            <TouchableWithoutFeedback onPress={() => setShowDatePicker(false)}>
              <View style={{ flex: 1, backgroundColor: 'rgba(0,0,0,0.72)', justifyContent: 'flex-end' }}>
                <TouchableWithoutFeedback>

                    <View style={[ui.row, { justifyContent: 'space-between', marginBottom: 16 }]}>
                      <Pressable accessibilityRole="button" accessibilityLabel="Previous month" onPress={() => {
                        const [year, month] = datePickerMonth.split('-').map(Number);
                        const previous = new Date(year, month - 2, 1);

                      <Text style={ui.heading}>{new Intl.DateTimeFormat(undefined, { month: 'long', year: 'numeric' }).format(new Date(`${datePickerMonth}-01T12:00:00`))}</Text>

                    </View>
                    <View style={{ flexDirection: 'row', justifyContent: 'space-around', marginBottom: 8 }}>
                      {['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((day, index) => <Text key={`${day}-${index}`} style={[ui.small, { width: '13%', textAlign: 'center' }]}>{day}</Text>)}

                      const cells: (number | null)[] = [...Array(offset).fill(null), ...Array.from({ length: count }, (_, index) => index + 1)];

                      const weeks = Array.from({ length: cells.length / 7 }, (_, index) => cells.slice(index * 7, index * 7 + 7));

                            if (day === null) return <View key={`blank-${weekIndex}-${dayIndex}`} style={{ width: '13%', height: 42 }} />;
                            const value = `${datePickerMonth}-${String(day).padStart(2, '0')}`;
                            const selected = value === newDate;
                            return (
                              <Pressable key={value} accessibilityRole="button" accessibilityLabel={value} accessibilityState={{ selected }}
                                onPress={() => { setNewDate(parseDateOnly(value)); setShowDatePicker(false); }}
                                style={{ width: '13%', height: 42, borderRadius: 12, alignItems: 'center', justifyContent: 'center', backgroundColor: selected ? '#76DCB9' : 'transparent' }}>
                                <Text style={[ui.text, selected ? { color: '#0C0E0D', fontWeight: '700' } : null]}>{day}</Text>

                          })}
                        </View>
                      ));
                    })()}
                    <Pressable accessibilityRole="button" onPress={() => { setNewDate(todayStr); setShowDatePicker(false); }} style={[ui.button, { alignSelf: 'center', marginTop: 8 }]}>
                      <Text style={ui.buttonText}>Today</Text>
                    </Pressable>
                  </View>
                </TouchableWithoutFeedback>
              </View>
            </TouchableWithoutFeedback>
          </Modal>

          <Text style={ui.small}>Amount</Text>
          <TextInput
            placeholder="0.00"
            placeholderTextColor="#AAB3AD"
            keyboardType="decimal-pad"
            value={newAmount}
            onChangeText={val => setNewAmount(acceptMoneyEdit(newAmount, val, currencyCode))}
            style={ui.input}
          />

          <SelectDropdown
            label="Category"
            sheetTitle="Select Transaction Category"
            value={newCategoryIdx}
            options={categoryDropdownOptions}
            onSelect={setNewCategoryIdx}
          />

          {CATEGORY_CHOICES[newCategoryIdx].isDebt && debts.length > 0 ? (
            <SelectDropdown
              label="Target Debt Account"
              sheetTitle="Select Target Debt Account"
            value={selectedDebtId}
              options={debtAccountDropdownOptions}
            onSelect={id => { setNewAccountId(id); setNewAppliesToDate(''); }}
            />
          ) : null}

          {CATEGORY_CHOICES[newCategoryIdx].category === 'payment' && selectedDebt && (selectedDebt.category === 'Mortgage' || selectedDebt.category === 'Auto loan') ? (
            <View style={{ marginTop: 8, padding: 10, backgroundColor: '#191E1B', borderRadius: 8, borderWidth: 1, borderColor: '#222A25' }}>
              <Text style={[ui.small, { color: '#76DCB9', fontWeight: 'bold' }]}>PAYMENT BREAKDOWN (OPTIONAL)</Text>
              <Text style={[ui.small, { color: '#AAB3AD', marginTop: 2, marginBottom: 8 }]}>
                If unknown, full payment is recorded in cashflow while loan principal remains unchanged.
              </Text>
              <Text style={ui.small}>Principal reduction</Text>
              <TextInput
                placeholder="0.00"
                placeholderTextColor="#AAB3AD"
                keyboardType="decimal-pad"
                value={newPrincipal}
                onChangeText={val => setNewPrincipal(acceptMoneyEdit(newPrincipal, val, currencyCode))}
                style={ui.input}
              />
              <Text style={[ui.small, { marginTop: 6 }]}>Interest paid</Text>
              <TextInput
                placeholder="0.00"
                placeholderTextColor="#AAB3AD"
                keyboardType="decimal-pad"
                value={newInterest}
                onChangeText={val => setNewInterest(acceptMoneyEdit(newInterest, val, currencyCode))}
                style={ui.input}
              />
              {selectedDebt.category === 'Mortgage' ? (
                <>
                  <Text style={[ui.small, { marginTop: 6 }]}>Escrow (taxes & insurance)</Text>
                  <TextInput
                    placeholder="0.00"
                    placeholderTextColor="#AAB3AD"
                    keyboardType="decimal-pad"
                    value={newEscrow}
                    onChangeText={val => setNewEscrow(acceptMoneyEdit(newEscrow, val, currencyCode))}
                    style={ui.input}
                  />
                </>
              ) : null}
              <Text style={[ui.small, { marginTop: 6 }]}>Fees</Text>
              <TextInput
                placeholder="0.00"
                placeholderTextColor="#AAB3AD"
                keyboardType="decimal-pad"
                value={newFee}
                onChangeText={val => setNewFee(acceptMoneyEdit(newFee, val, currencyCode))}
                style={ui.input}
              />
            </View>
          ) : null}

          <Pressable
            accessibilityRole="button"
            disabled={savingTx}
            onPress={handleSaveTransaction}
            style={[ui.button, { backgroundColor: '#76DCB9', marginTop: 12, justifyContent: 'center' }]}
          >
            <Text style={[ui.buttonText, { color: '#0C0E0D', fontWeight: 'bold' }]}>
              {savingTx ? 'Saving\u2026' : 'Save Transaction'}
            </Text>
          </Pressable>
        </Card>
      ) : null}

      {tab === 'Upcoming' ? (
        <Card>
          <Text style={ui.heading}>Upcoming and overdue</Text>
          <Text style={ui.text}>Unpaid obligations due this month, plus genuine unpaid overdue cycles from earlier months.</Text>
          {!!billError ? <Text accessibilityRole="alert" style={ui.text}>{billError}</Text> : null}
          {(() => {
            const rows = getScheduledObligationRows({ debts, recurringBills, transactions, asOfDate: todayStr, currencyCode, schedules: obligationSchedules, loanBillPairs })
              .filter(row => row.state.status !== 'paid' && (row.dueDate.slice(0, 7) === currentMonthStr || row.dueDate < todayStr));
            const grouped = new Map<string, typeof rows>();
            for (const row of rows) {
              const month = row.dueCycleMonth;
              grouped.set(month, [...(grouped.get(month) ?? []), row]);
            }
            if (!rows.length) return <Text style={[ui.text, { marginTop: 8 }]}>Nothing due this month or overdue.</Text>;
            return [...grouped.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([month, monthRows]) => (
              <View key={month} style={{ marginTop: 12 }}>
                <Text style={[ui.heading, { marginBottom: 4 }]}>{new Intl.DateTimeFormat(undefined, { month: 'long', year: 'numeric' }).format(new Date(`${month}-01T12:00:00`))}</Text>
                {monthRows.map(row => {
                  const { bill, debt, dueDate: due, state } = row;
                  const match = state.confirmedMatch;
                  const pending = state.status === 'pending_confirmation';
                  const pastDue = due < todayStr;
                  return <View key={row.id} style={[ui.divider, ui.row, { justifyContent: 'space-between', gap: 8 }]}>
                    <View style={{ flex: 1 }}><Text style={ui.heading}>{row.kind === 'debt' ? `${debt?.name ?? row.name} minimum` : bill?.name ?? row.name}</Text><Text style={[ui.small, pastDue ? { color: '#F08B78' } : null]}>{`${pastDue ? 'Past due since' : 'Due'} ${due} · ${match.isPartiallyPaid ? 'Partially paid' : pending ? 'Awaiting confirmation' : 'Unpaid'}`}</Text></View>
                    <View style={{ alignItems: 'flex-end', gap: 5 }}><Text style={ui.value}>{formatCurrency(moneyToMajor(match.remainingAmount), currencySymbol)}</Text><Pressable accessibilityRole="button" accessibilityLabel={`Open payment details for ${row.name}`} disabled={pending} onPress={() => { setUpcomingError(''); setUpcomingTarget(row); }} style={[ui.button, { backgroundColor: '#76DCB9' }]}><Text style={[ui.buttonText, { color: '#0C0E0D' }]}>{pending ? 'Awaiting confirmation' : 'Pay now'}</Text></Pressable></View>
                  </View>;
                })}
              </View>
            ));
          })()}
        </Card>
      ) : tab === 'Scheduled' ? (
        <Card>
          <Text style={ui.heading}>Scheduled obligations</Text>
          <Text style={ui.text}>Browse monthly recurring bills and debt minimums for the next 12 months. Payments are recorded today and linked to the due date.</Text>
          {(() => {
            let selectedMonth = currentMonthStr;
            for (let index = 0; index < scheduledMonthOffset; index += 1) selectedMonth = getNextMonthString(selectedMonth);
            return <View style={[ui.row, { justifyContent: 'space-between', alignItems: 'center', marginTop: 10 }]}>
              <Pressable accessibilityRole="button" accessibilityLabel="Previous scheduled month" disabled={scheduledMonthOffset === 0} onPress={() => setScheduledMonthOffset(value => Math.max(0, value - 1))} style={[ui.button, scheduledMonthOffset === 0 && { opacity: 0.45 }]}><Text style={ui.buttonText}>Previous</Text></Pressable>
              <Text accessibilityRole="header" style={ui.heading}>{new Intl.DateTimeFormat(undefined, { month: 'long', year: 'numeric' }).format(new Date(`${selectedMonth}-01T12:00:00`))}</Text>
              <Pressable accessibilityRole="button" accessibilityLabel="Next scheduled month" disabled={scheduledMonthOffset >= 11} onPress={() => setScheduledMonthOffset(value => Math.min(11, value + 1))} style={[ui.button, scheduledMonthOffset >= 11 && { opacity: 0.45 }]}><Text style={ui.buttonText}>Next</Text></Pressable>
            </View>;
          })()}
          {!!billError ? <Text accessibilityRole="alert" style={ui.text}>{billError}</Text> : null}
          {(() => {
            const allRows = getScheduledObligationRows({
              debts,
              recurringBills,
              transactions,
              asOfDate: todayStr,
              currencyCode,
              schedules: obligationSchedules,
              loanBillPairs,
            });
            let selectedMonth = currentMonthStr;
            for (let index = 0; index < scheduledMonthOffset; index += 1) selectedMonth = getNextMonthString(selectedMonth);
            const visibleRows = allRows.filter(row => row.dueCycleMonth === selectedMonth || (scheduledMonthOffset === 0 && row.dueDate < todayStr));
            if (!visibleRows.length) return <Text style={[ui.text, { marginTop: 8 }]}>No scheduled obligations for this month.</Text>;
            const grouped = new Map<string, typeof visibleRows>();
            for (const row of visibleRows) grouped.set(row.dueCycleMonth, [...(grouped.get(row.dueCycleMonth) ?? []), row]);
            return [...grouped.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([month, monthRows]) => <View key={month} style={{ marginTop: 8 }}>

              {monthRows.map(row => {
              const { bill, debt, dueDate: due, state } = row;
              const match = state.confirmedMatch;
              const isPending = state.status === 'pending_confirmation';
              const isPastDue = due < todayStr && state.status !== 'paid';
              const busyKey = `${row.kind}:${row.kind === 'debt' ? debt?.id : bill?.id}:${due}`;
              const displayName = row.kind === 'debt' ? `${debt?.name ?? row.name} minimum` : bill?.name ?? row.name;
              const recordPayment = async () => {
                setPendingBillCycle(busyKey);
                setBillError('');
                try {
                  if (row.kind === 'debt' && debt && onAddTransaction) {
                    await onAddTransaction(createDebtMinimumPaymentTransaction({
                      id: `tx:${Crypto.randomUUID()}`, debt, today: todayStr, dueDate: due,
                      amount: match.remainingAmount, createdAt: new Date().toISOString(),
                    }));
                  } else if (row.kind === 'bill' && bill) {
                    await onRecordBillPayment?.(bill, due, match.remainingAmount);
                  }
                } catch (error) {
                  setBillError(error instanceof Error ? error.message : 'Unable to record this payment.');
                } finally {
                  setPendingBillCycle(null);
                }
              };
              return (
              <View key={row.id} style={[ui.divider, ui.row, { justifyContent: 'space-between', gap: 8 }]}>
                <View style={{ flex: 1 }}>
                  <Text style={ui.heading}>{displayName}</Text>
                  <Text style={ui.small}>{`${isPastDue ? 'Past due since ' : 'Due '}${due} · ${row.kind === 'debt' ? 'Minimum payment' : bill?.paymentMethod === 'credit' ? 'Charged to credit card' : 'Paid from checking'}`}</Text>
                  <Text style={[ui.small, { color: match.isSatisfied ? '#86E3B8' : isPending ? '#F5C563' : isPastDue ? '#F08B78' : '#AAB3AD' }]}>
                    {match.isSatisfied
                      ? 'Paid'
                      : isPending
                        ? `Awaiting confirmation · ${formatCurrency(moneyToMajor(state.pendingAmount), currencySymbol)} pending`
                        : match.isPartiallyPaid
                          ? `Partially paid · ${formatCurrency(moneyToMajor(match.remainingAmount), currencySymbol)} left`
                          : isPastDue ? 'Past due · Unpaid' : 'Unpaid'}
                  </Text>
                </View>
                <View style={{ alignItems: 'flex-end', gap: 6 }}>
                  <Text style={ui.value}>{formatCurrency(moneyToMajor(row.requiredAmount), currencySymbol)}</Text>
                  <Pressable accessibilityRole="button" disabled={(row.kind === 'debt' ? !onAddTransaction : !onRecordBillPayment) || match.isSatisfied || isPending || pendingBillCycle === busyKey}
                    onPress={recordPayment} style={[ui.button, { backgroundColor: '#76DCB9' }]}>
                    <Text style={[ui.buttonText, { color: '#0C0E0D' }]}>{pendingBillCycle === busyKey ? 'Saving…' : match.isSatisfied ? 'Paid' : isPending ? 'Awaiting confirmation' : match.isPartiallyPaid ? 'Record remaining' : 'Record paid'}</Text>
                  </Pressable>
                </View>
              </View>
              );
              })}
            </View>);
          })()}
        </Card>
      ) : (
        <>
          <View style={[ui.row, { gap: 8, marginVertical: 4 }]}>
            <View style={{ flex: 1 }}>
              <SelectDropdown
                sheetTitle="Filter by Period"
                value={periodIdx}
                options={periodDropdownOptions}
                onSelect={setPeriodIdx}
              />
            </View>
            <View style={{ flex: 1 }}>
              <SelectDropdown
                sheetTitle="Filter by Account"
                value={accountIdx}
                options={accountFilterDropdownOptions}
                onSelect={setAccountIdx}
              />
            </View>
          </View>

          <View style={ui.row}>
            <View style={[ui.card, { flex: 1, padding: 10, backgroundColor: '#141816', borderRadius: 12, borderWidth: 1, borderColor: '#222A25' }]}>
              <View style={[ui.row, { gap: 5 }]}>
                <AppIcon name="trendDown" size={13} color="#76DCB9" />
                <Text numberOfLines={1} style={ui.small}>Payments</Text>
              </View>
              <Text style={ui.heading}>{formatCurrency(moneyToMajor(makeMoney(currencyCode, totalPaymentsMinor)), currencySymbol)}</Text>
            </View>
            <View style={[ui.card, { flex: 1, padding: 10, backgroundColor: '#141816', borderRadius: 12, borderWidth: 1, borderColor: '#222A25' }]}>
              <View style={[ui.row, { gap: 5 }]}>
                <AppIcon name="brandRise" size={13} color="#F08B78" />
                <Text numberOfLines={1} style={ui.small}>Interest & fees</Text>
              </View>
              <Text style={ui.heading}>{formatCurrency(moneyToMajor(makeMoney(currencyCode, totalInterestFeesMinor)), currencySymbol)}</Text>
            </View>
            <View style={[ui.card, { flex: 1, padding: 10, backgroundColor: '#141816', borderRadius: 12, borderWidth: 1, borderColor: '#222A25' }]}>
              <View style={[ui.row, { gap: 5 }]}>
                <AppIcon name="sort" size={13} color="#AAB3AD" />
                <Text numberOfLines={1} style={ui.small}>Adjustments</Text>
              </View>
              <Text style={ui.heading}>{formatCurrency(moneyToMajor(makeMoney(currencyCode, totalAdjustmentsMinor)), currencySymbol)}</Text>
            </View>
          </View>

          <Card>
            <View style={[ui.row, { justifyContent: 'space-between' }]}>
              <View style={[ui.row, { gap: 6 }]}>
                <Text style={ui.text}>Net balance change</Text>
                <AppIcon name="info" size={15} color="#AAB3AD" />
              </View>
              <Text
                style={[
                  ui.value,
                  { color: netBalanceChangeMinor < 0 ? '#76DCB9' : netBalanceChangeMinor > 0 ? '#F08B78' : '#F9F9F6' },
                ]}
              >
                {netBalanceChangeMinor < 0
                  ? `-${formatCurrency(moneyToMajor(makeMoney(currencyCode, Math.abs(netBalanceChangeMinor))), currencySymbol)}`
                  : netBalanceChangeMinor > 0
                    ? `+${formatCurrency(moneyToMajor(makeMoney(currencyCode, netBalanceChangeMinor)), currencySymbol)}`
                    : formatCurrency(0, currencySymbol)}
              </Text>
            </View>
          </Card>

          <View style={[ui.row, { gap: 8 }]}>
            <View style={[ui.inputWrap, { flex: 1 }]}>
              <AppIcon name="search" size={16} color="#A6B2AA" />
              <TextInput
                accessibilityLabel="Search transactions"
                placeholder="Search payee or category"
                placeholderTextColor="#77847D"
                value={search}
                onChangeText={setSearch}
                style={[ui.input, { borderWidth: 0, paddingHorizontal: 0, paddingVertical: 8, flex: 1 }]}
              />
            </View>
            <Pressable
              accessibilityRole="button"
              accessibilityLabel={sortAsc ? 'Sort newest first' : 'Sort oldest first'}
              onPress={() => setSortAsc(prev => !prev)}
              style={[ui.button, { width: 44, height: 42, alignItems: 'center', justifyContent: 'center', paddingHorizontal: 0 }]}
            >
              <Text style={{ color: '#76DCB9', fontSize: 16, fontWeight: '700' }}>⇅</Text>
            </Pressable>
          </View>
          <SelectDropdown value={typeIdx} options={typeDropdownOptions} onSelect={setTypeIdx} sheetTitle="Filter transaction type" />

          {sorted.length === 0 ? (
            <Card>
              <Text style={ui.heading}>No recorded transactions</Text>
              <Text style={ui.text}>
                No transactions match your current search and filters. Tap the + button above to log a payment or purchase.
              </Text>
            </Card>
          ) : (
            sorted.map(tx => {
              const dir = classifyCategoryDirection(tx.category);
              const targetDebt = debts.find(d => d.id === tx.accountId);
              const accountLabel = tx.accountId === null ? 'Checking / Debit' : targetDebt?.name || 'Debt Account';
              const isPayment = dir === 'decrease_debt';
              const isCharge = dir === 'increase_debt';
              const amountFormatted = formatCurrency(moneyToMajor(tx.amount), currencySymbol);

              return (
                <Pressable
                  key={tx.id}
                  accessibilityRole="button"
                  accessibilityLabel={`Edit transaction ${tx.payee}`}
                  onPress={() => handleOpenEdit(tx)}
                >
                  <Card>
                    <View style={[ui.row, { justifyContent: 'space-between' }]}>
                      <View style={[ui.row, { gap: 10, flex: 1 }]}>
                        <View style={ui.iconCircle}>
                          <AppIcon
                            name={
                              isPayment
                                ? 'trendDown'
                                : tx.category === 'interest' || tx.category === 'fee'
                                  ? 'brandRise'
                                  : tx.accountId === null
                                    ? 'calendar'
                                    : 'creditCard'
                            }
                            size={18}
                            color={isPayment ? '#86E3B8' : isCharge ? '#F08B78' : '#76DCB9'}
                          />
                        </View>
                        <View style={{ flex: 1 }}>
                          <Text style={ui.heading}>{tx.payee}</Text>
                          <Text style={ui.small}>
                            {`${tx.date} \u00b7 ${accountLabel} \u00b7 ${tx.category.replace('_', ' ')}`}
                          </Text>
                          {tx.appliesToDate ? <Text style={ui.small}>{`Applies to due cycle ${tx.appliesToDate}`}</Text> : null}
                        </View>
                      </View>
                      <View style={{ alignItems: 'flex-end' }}>
                        <View style={[ui.row, { gap: 6, alignItems: 'center' }]}>
                          <Text
                            style={[
                              ui.value,
                              { color: isPayment ? '#86E3B8' : isCharge ? '#F08B78' : '#F1F5F2' },
                            ]}
                          >
                            {isPayment ? `-${amountFormatted}` : isCharge ? `+${amountFormatted}` : amountFormatted}
                          </Text>
                          <AppIcon name="chevronRight" size={14} color="#AAB3AD" />
                        </View>
                        {tx.isPending ? (
                          <View style={[ui.badge, { backgroundColor: '#38321E', borderColor: '#F5C563', borderWidth: 1 }]}>
                            <Text style={[ui.badgeText, { color: '#F5C563', fontSize: 10 }]}>Pending Confirmation</Text>
                          </View>
                        ) : tx.source === 'recurring_bill' ? (
                          <View style={[ui.badge, { backgroundColor: '#1E2C28' }]}>
                            <Text style={[ui.badgeText, { color: '#76DCB9', fontSize: 10 }]}>Auto-Bill</Text>
                          </View>
                        ) : tx.source === 'reconciliation' ? (
                          <View style={[ui.badge, { backgroundColor: '#2D2825' }]}>
                            <Text style={[ui.badgeText, { color: '#F08B78', fontSize: 10 }]}>Reconciled</Text>
                          </View>
                        ) : null}
                      </View>
                    </View>
                    {tx.memo ? (
                      <Text style={[ui.small, { marginTop: 4, fontStyle: 'italic' }]}>{tx.memo}</Text>
                    ) : null}
                  </Card>
                </Pressable>
              );
            })
          )}
        </>
      )}

      <Modal visible={upcomingTarget !== null} transparent animationType="slide" onRequestClose={() => setUpcomingTarget(null)}>
        <View style={{ flex: 1, backgroundColor: 'rgba(0,0,0,0.72)', justifyContent: 'flex-end' }}>
          <View accessibilityViewIsModal style={{ backgroundColor: '#141816', borderTopLeftRadius: 20, borderTopRightRadius: 20, borderWidth: 1, borderColor: '#222A25', padding: 18, maxHeight: '90%' }}>
            <View style={[ui.row, { justifyContent: 'space-between', marginBottom: 12 }]}>
              <View style={{ flex: 1 }}><Text style={ui.title}>{`Payment for ${upcomingTarget?.name ?? ''}`}</Text><Text style={ui.small}>{`Due ${upcomingTarget?.dueDate ?? ''}`}</Text></View>
              <Pressable accessibilityRole="button" accessibilityLabel="Close payment details" onPress={() => setUpcomingTarget(null)} style={ui.button}><Text style={ui.buttonText}>X</Text></Pressable>
            </View>
            {!!upcomingError && <Text accessibilityRole="alert" style={[ui.small, { color: '#F08B78', marginBottom: 8 }]}>{upcomingError}</Text>}
            <View style={[ui.card, { padding: 12, marginBottom: 12 }]}>
              <View style={[ui.row, { justifyContent: 'space-between' }]}><View><Text style={ui.small}>Required</Text><Text style={ui.heading}>{formatCurrency(moneyToMajor(upcomingTarget?.requiredAmount ?? makeMoney(currencyCode, 0)), currencySymbol)}</Text></View><View><Text style={ui.small}>Confirmed paid</Text><Text style={ui.heading}>{formatCurrency(moneyToMajor(upcomingPaymentState?.confirmedMatch.totalPaid ?? makeMoney(currencyCode, 0)), currencySymbol)}</Text></View><View><Text style={ui.small}>Remaining</Text><Text style={ui.value}>{formatCurrency(moneyToMajor(upcomingPaymentState?.confirmedMatch.remainingAmount ?? makeMoney(currencyCode, 0)), currencySymbol)}</Text></View></View>
              {upcomingPaymentState?.pendingAmount.minorUnits ? <Text style={[ui.small, { color: '#F5C563' }]}>{`${formatCurrency(moneyToMajor(upcomingPaymentState.pendingAmount), currencySymbol)} awaiting confirmation`}</Text> : null}
            </View>
            <ScrollView style={{ maxHeight: 290 }} keyboardShouldPersistTaps="handled">
              <Text style={[ui.heading, { fontSize: 15, marginBottom: 4 }]}>Match a recorded payment</Text>
              <Text style={[ui.small, { marginBottom: 8 }]}>Link an existing unassigned payment to this due cycle.</Text>
              {upcomingCandidates.length ? upcomingCandidates.map(tx => <View key={tx.id} style={[ui.divider, ui.row, { justifyContent: 'space-between' }]}>
                <View style={{ flex: 1 }}><Text style={ui.heading}>{tx.payee}</Text><Text style={ui.small}>{`${tx.date} · ${formatCurrency(moneyToMajor(tx.amount), currencySymbol)}`}</Text></View>
                <Pressable accessibilityRole="button" accessibilityLabel={`Match payment from ${tx.payee}`} disabled={upcomingBusy || !onUpdateTransaction} onPress={() => void matchUpcomingPayment(tx)} style={ui.button}><Text style={ui.buttonText}>Match</Text></Pressable>
              </View>) : <Text style={ui.small}>No unassigned payments are available to match.</Text>}
            </ScrollView>
            <Pressable accessibilityRole="button" accessibilityLabel="Record a new payment" disabled={upcomingBusy || !upcomingPaymentState || upcomingPaymentState.status === 'pending_confirmation' || upcomingPaymentState.confirmedMatch.remainingAmount.minorUnits <= 0} onPress={() => void recordUpcomingPayment()} style={[ui.button, { alignItems: 'center', marginTop: 12, backgroundColor: '#76DCB9' }]}>
              <Text style={[ui.buttonText, { color: '#0C0E0D' }]}>{upcomingBusy ? 'Saving…' : `Record new payment${upcomingPaymentState ? ` (${formatCurrency(moneyToMajor(upcomingPaymentState.confirmedMatch.remainingAmount), currencySymbol)})` : ''}`}</Text>
            </Pressable>
          </View>
        </View>
      </Modal>

      {/* Edit Transaction Modal */}
      <Modal visible={editingTx !== null} transparent animationType="slide" onRequestClose={() => { setEditingTx(null); onEditDismiss?.(); }}>
        <View style={{ flex: 1, backgroundColor: 'rgba(0,0,0,0.72)', justifyContent: 'flex-end' }}>
          <View style={{ backgroundColor: '#141816', borderTopLeftRadius: 20, borderTopRightRadius: 20, borderWidth: 1, borderColor: '#222A25', padding: 18, maxHeight: '90%' }}>
            <ScrollView showsVerticalScrollIndicator={false}>
              <View style={[ui.row, { justifyContent: 'space-between', marginBottom: 12 }]}>
                <Text style={ui.title}>Edit Transaction</Text>
                <Pressable
                  accessibilityRole="button"
                  accessibilityLabel="Close edit form"
                  onPress={() => { setEditingTx(null); onEditDismiss?.(); }}
                >
                  <Text style={{ color: '#AAB3AD', fontSize: 18, fontWeight: 'bold' }}>X</Text>
                </Pressable>
              </View>

              {!!editError && (
                <Text accessibilityRole="alert" style={[ui.small, { color: '#F08B78', marginBottom: 8 }]}>
                  {editError}
                </Text>
              )}

              {editingTx?.isPending ? (
                <View style={{ backgroundColor: '#2D281E', borderColor: '#F5C563', borderWidth: 1, borderRadius: 8, padding: 8, marginBottom: 8 }}>
                  <Text style={[ui.small, { color: '#F5C563', fontWeight: 'bold' }]}>Pending Confirmation</Text>
                  <Text style={[ui.small, { color: '#D4C9A8' }]}>This transaction has not yet been confirmed by the server.</Text>
                </View>
              ) : null}

              <Text style={ui.small}>Payee or Description</Text>
              <TextInput
                accessibilityLabel="Payee or description"
                placeholder="e.g. Target, Payment"
                placeholderTextColor="#AAB3AD"
                value={editPayee}
                onChangeText={setEditPayee}
                style={ui.input}
              />

              <Text style={[ui.small, { marginTop: 8 }]}>Amount</Text>
              <TextInput
                accessibilityLabel="Transaction amount"
                placeholder="0.00"
                placeholderTextColor="#AAB3AD"
                keyboardType="decimal-pad"
                value={editAmount}
                onChangeText={val => setEditAmount(acceptMoneyEdit(editAmount, val, currencyCode))}
                style={ui.input}
              />

              <SelectDropdown
                label="Category"
                sheetTitle="Select Transaction Category"
                value={editCategoryIdx}
                options={categoryDropdownOptions}
                onSelect={setEditCategoryIdx}
              />

              {editCatOption.isDebt && debts.length > 0 ? (
                <SelectDropdown
                  label="Target Debt Account"
                  sheetTitle="Select Target Debt Account"
                  value={editAccountId || debts[0]?.id || ''}
                  options={debtAccountDropdownOptions}
                  onSelect={setEditAccountId}
                />
              ) : null}

              {editCatOption.category === 'payment' && targetEditDebt && (targetEditDebt.category === 'Mortgage' || targetEditDebt.category === 'Auto loan') ? (
                <View style={{ marginTop: 8, padding: 10, backgroundColor: '#191E1B', borderRadius: 8, borderWidth: 1, borderColor: '#222A25' }}>
                  <Text style={[ui.small, { color: '#76DCB9', fontWeight: 'bold' }]}>PAYMENT BREAKDOWN (OPTIONAL)</Text>
                  <Text style={[ui.small, { color: '#AAB3AD', marginTop: 2, marginBottom: 8 }]}>
                    If unknown, full payment is recorded in cashflow while loan principal remains unchanged.
                  </Text>
                  <Text style={ui.small}>Principal reduction</Text>
                  <TextInput
                    placeholder="0.00"
                    placeholderTextColor="#AAB3AD"
                    keyboardType="decimal-pad"
                    value={editPrincipal}
                    onChangeText={val => setEditPrincipal(acceptMoneyEdit(editPrincipal, val, currencyCode))}
                    style={ui.input}
                  />
                  <Text style={[ui.small, { marginTop: 6 }]}>Interest paid</Text>
                  <TextInput
                    placeholder="0.00"
                    placeholderTextColor="#AAB3AD"
                    keyboardType="decimal-pad"
                    value={editInterest}
                    onChangeText={val => setEditInterest(acceptMoneyEdit(editInterest, val, currencyCode))}
                    style={ui.input}
                  />
                  {targetEditDebt.category === 'Mortgage' ? (
                    <>
                      <Text style={[ui.small, { marginTop: 6 }]}>Escrow (taxes & insurance)</Text>
                      <TextInput
                        placeholder="0.00"
                        placeholderTextColor="#AAB3AD"
                        keyboardType="decimal-pad"
                        value={editEscrow}
                        onChangeText={val => setEditEscrow(acceptMoneyEdit(editEscrow, val, currencyCode))}
                        style={ui.input}
                      />
                    </>
                  ) : null}
                  <Text style={[ui.small, { marginTop: 6 }]}>Fees</Text>
                  <TextInput
                    placeholder="0.00"
                    placeholderTextColor="#AAB3AD"
                    keyboardType="decimal-pad"
                    value={editFee}
                    onChangeText={val => setEditFee(acceptMoneyEdit(editFee, val, currencyCode))}
                    style={ui.input}
                  />
                </View>
              ) : null}

              <Text style={[ui.small, { marginTop: 8 }]}>Transaction Date (YYYY-MM-DD)</Text>
              <TextInput
                accessibilityLabel="Transaction date"
                placeholder="YYYY-MM-DD"
                placeholderTextColor="#AAB3AD"
                value={editDate}
                onChangeText={setEditDate}
                style={ui.input}
              />

              <Text style={[ui.small, { marginTop: 8 }]}>Note (optional)</Text>
              <TextInput
                accessibilityLabel="Transaction note"
                placeholder="Optional notes or memo"
                placeholderTextColor="#AAB3AD"
                value={editMemo}
                onChangeText={setEditMemo}
                style={ui.input}
              />

              {/* Live Balance Preview */}
              <View style={[ui.card, { backgroundColor: '#262927', padding: 12, marginVertical: 12, borderRadius: 8, borderWidth: 1, borderColor: '#353A37' }]}>
                <Text style={[ui.small, { color: '#76DCB9', fontWeight: 'bold' }]}>LIVE BALANCE PREVIEW</Text>
                {targetEditDebt ? (
                  <View style={[ui.row, { justifyContent: 'space-between', marginTop: 6 }]}>
                    <Text style={ui.heading}>{targetEditDebt.name}</Text>
                    <Text style={[ui.value, { fontSize: 13 }]}>
                      {formatCurrency(editCurrentBal, currencySymbol)} â†’ <Text style={{ color: editAfterBal <= editCurrentBal ? '#76DCB9' : '#F08B78' }}>{formatCurrency(editAfterBal, currencySymbol)}</Text>
                    </Text>
                  </View>
                ) : (
                  <Text style={[ui.text, { marginTop: 4 }]}>Checking / Cashflow ($0 debt impact)</Text>
                )}
              </View>

              {/* Actions */}
              <View style={{ gap: 10, marginTop: 4, marginBottom: 20 }}>
                <Pressable
                  accessibilityRole="button"
                  accessibilityLabel="Save Changes"
                  disabled={savingEdit}
                  onPress={() => void handleSaveEdit()}
                  style={[ui.button, { backgroundColor: '#76DCB9', justifyContent: 'center', minHeight: 44 }]}
                >
                  <Text style={[ui.buttonText, { color: '#0C0E0D', fontWeight: 'bold' }]}>
                    {savingEdit ? 'Savingâ€¦' : 'Save Changes'}
                  </Text>
                </Pressable>

                <Pressable
                  accessibilityRole="button"
                  accessibilityLabel="Cancel edit"
                  onPress={() => { setEditingTx(null); onEditDismiss?.(); }}
                  style={[ui.button, { justifyContent: 'center', minHeight: 40 }]}
                >
                  <Text style={ui.buttonText}>Cancel</Text>
                </Pressable>

                <Pressable
                  accessibilityRole="button"
                  accessibilityLabel="Delete transaction"
                  disabled={deletingTx}
                  onPress={() => void handleDeleteTx()}
                  style={[ui.button, { backgroundColor: '#3D2520', borderColor: '#8B4F42', borderWidth: 1, justifyContent: 'center', minHeight: 44 }]}
                >
                  <Text style={[ui.buttonText, { color: '#F08B78', fontWeight: 'bold' }]}>
                    {deletingTx ? 'Deletingâ€¦' : 'Delete transaction'}
                  </Text>
                </Pressable>
              </View>
            </ScrollView>
          </View>
        </View>
      </Modal>
    </FinancialScreen>
  );
}
